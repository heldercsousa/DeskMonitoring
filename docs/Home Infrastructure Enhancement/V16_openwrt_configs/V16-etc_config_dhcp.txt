# =====================================================================
# /etc/config/dhcp  —  V16  (dnsmasq: DHCP + resolvedor local por VLAN)
# =====================================================================

config dnsmasq
	option domainneeded '1'
	option localise_queries '1'
	option rebind_protection '1'      # anti DNS-rebinding
	option rebind_localhost '1'
	option local '/lan/'
	option domain 'lan'
	option expandhosts '1'
	option cachesize '1000'
	option authoritative '1'
	option readethers '1'
	option localservice '1'
	# Upstream seguro (Quad9 bloqueia dominios maliciosos). Use DoT/DoH se instalar
	# AdGuard Home ou https-dns-proxy e aponte aqui p/ 127.0.0.1#5053.
	list server '9.9.9.9'
	list server '149.112.112.112'

config dhcp 'vlan30'
	option interface 'vlan30'
	option start '100'
	option limit '150'
	option leasetime '12h'

config dhcp 'vlan20'
	option interface 'vlan20'
	option start '100'
	option limit '150'
	option leasetime '12h'

config dhcp 'vlan40'
	option interface 'vlan40'
	option start '100'
	option limit '150'
	option leasetime '12h'

config dhcp 'vlan50'
	option interface 'vlan50'
	option start '100'
	option limit '150'
	option leasetime '4h'

# WAN nunca serve DHCP
config dhcp 'wan'
	option interface 'wan'
	option ignore '1'

config odhcpd 'odhcpd'
	option maindhcp '0'
	option leasefile '/tmp/hosts/odhcpd'
	option leasetrigger '/usr/sbin/odhcpd-update'
	option loglevel '4'
