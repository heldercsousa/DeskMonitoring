# =====================================================================
# /etc/config/network  —  V16 Zero-Trust (Xiaomi AX3000T / OpenWrt 24.10 DSA)
# WAN = HGU Vivo em BRIDGE -> porta WAN. PPPoE + VLAN 10 aqui.
# Gestao (SSH/LuCI) vem pela vlan30. Esteja na porta lan1 (vlan30) ao aplicar.
# =====================================================================

config interface 'loopback'
	option device 'lo'
	option proto 'static'
	option ipaddr '127.0.0.1'
	option netmask '255.0.0.0'

config globals 'globals'
	option ula_prefix 'fd00:2b:2b::/48'

# ---------- Bridge fisica + Bridge-VLAN filtering (DSA) ----------
config device
	option name 'br-lan'
	option type 'bridge'
	list ports 'lan1'
	list ports 'lan2'
	list ports 'lan3'
	list ports 'lan4'

config bridge-vlan
	option device 'br-lan'
	option vlan '20'
	list ports 'lan4:u*'      # WORK-DEV  (porta 4 untagged)

config bridge-vlan
	option device 'br-lan'
	option vlan '30'
	list ports 'lan1:u*'      # HOME-MAIN / GESTAO (porta 1 untagged)

config bridge-vlan
	option device 'br-lan'
	option vlan '40'
	list ports 'lan2:u*'      # HOME-IOT  (porta 2 untagged)

config bridge-vlan
	option device 'br-lan'
	option vlan '50'
	list ports 'lan3:u*'      # GUEST-WIFI (porta 3 untagged)

# ---------- Interfaces logicas por VLAN (IP + gateway do segmento) ----------
config interface 'vlan20'
	option device 'br-lan.20'
	option proto 'static'
	option ipaddr '192.168.20.1'
	option netmask '255.255.255.0'

config interface 'vlan30'
	option device 'br-lan.30'
	option proto 'static'
	option ipaddr '192.168.30.1'
	option netmask '255.255.255.0'

config interface 'vlan40'
	option device 'br-lan.40'
	option proto 'static'
	option ipaddr '192.168.40.1'
	option netmask '255.255.255.0'

config interface 'vlan50'
	option device 'br-lan.50'
	option proto 'static'
	option ipaddr '192.168.50.1'
	option netmask '255.255.255.0'

# ---------- WAN: VLAN 10 sobre a porta 'wan' + PPPoE Vivo ----------
config device
	option name 'wan.10'
	option type '8021q'
	option ifname 'wan'
	option vid '10'

config interface 'wan'
	option device 'wan.10'
	option proto 'pppoe'
	option username 'cliente@cliente'
	option password 'cliente'
	option mtu '1492'
	option ipv6 'auto'

config interface 'wan6'
	option device 'wan.10'
	option proto 'dhcpv6'
